async function initPage() { try { const resMovies = await fetch('/api/movies'); if (resMovies.ok) { const movies = await resMovies.json(); const movie = movies.find(m => m.title.toLowerCase() === movieTitleParam.toLowerCase()); if (movie) { if (movie.price) ticketPrice = Number(movie.price); // Jika admin mengatur jadwal khusus, render tombol jam tayangnya secara dinamis if (movie.schedule) { const times = movie.schedule.split(',').map(t => t.trim()).filter(Boolean); if (times.length > 0) { selectedTime = times[0]; // Set default ke jadwal pertama const timeContainer = document.getElementById('timeSlotsContainer'); timeContainer.innerHTML = times.map((t, index) => ` `).join(''); } } } } const resBookings = await fetch('/api/bookings'); if (resBookings.ok) { const bookings = await resBookings.json(); bookings.forEach(b => { if (b.movie_title && b.movie_title.toLowerCase().includes(movieTitleParam.toLowerCase())) { if (b.seats) { const seatsArr = b.seats.split(',').map(s => s.trim()); bookedSeatsList.push(...seatsArr); } } }); } } catch (err) { console.error("Gagal memuat detail data:", err); } renderSeats(); }
